let recognition; let timerInatividade; // Verifica se o navegador suporta Web Speech API const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition; if (SpeechRecognition) { recognition = new SpeechRecognition(); recognition.continuous = true; // Permite continuar ouvindo dentro da janela de tempo recognition.interimResults = true; // Captura o áudio enquanto a pessoa fala // Quando o usuário começa a falar e o sistema detecta palavras recognition.onresult = (event) => { let textoTranscrito = ''; for (let i = event.resultIndex; i < event.results.length; i++) { textoTranscrito += event.results[i][0].transcript; } console.log("Texto detectado:", textoTranscrito); // Reinicia o temporizador de 10 segundos toda vez que uma palavra/frase chega reiniciarTimerSilencio(); }; // Quando o microfone é desligado (manualmente ou por timeout) recognition.onend = () => { console.log("Microfone desligado. Aguardando novo clique para reativar."); limparTimer(); }; recognition.onerror = (event) => { console.error("Erro no microfone:", event.error); limparTimer(); }; } // Função para iniciar a escuta SOMENTE no clique function ligarMicrofoneManualmente() { if (!recognition) return; limparTimer(); recognition.start(); console.log("Microfone ativado por clique. Fale a primeira palavra..."); } // Controla os 10 segundos de espera após ouvir a fala function reiniciarTimerSilencio() { limparTimer(); // Espera 10 segundos (10000 ms) sem novas palavras para desligar timerInatividade = setTimeout(() => { console.log("10 segundos de silêncio passados. Desligando microfone..."); recognition.stop(); }, 10000); } function limparTimer() { if (timerInatividade) { clearTimeout(timerInatividade); timerInatividade = null; } }
